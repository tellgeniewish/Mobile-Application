            .setSmallIcon(R.drawable.ic_stat_name)
            .setContentTitle("알람!!!")
            .setContentText(alarmMessage)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)

        val notiManager = NotificationManagerCompat.from(context)
        notiManager.notify(100, alarmNotiBuilder.build())
